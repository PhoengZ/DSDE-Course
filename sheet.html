3.1 Build class for pipeline

from sklearn.base import BaseEstimator, TransformerMixin
class DropWMissing(BaseEstimator, TransformerMixin):
    def __init__(self, threshold=0.5):
        self.threshold = threshold
    def fit(self, X, y=None):
        X = X.copy()
        self.featurein_ = X.columns.tolist()
        vals = X.isnull().sum()
        vals = vals[vals > self.threshold*X.shape[0]]
        self.remove_cols_ = vals.index.tolist()
        return self
    def transform(self, X):
        X = X.copy()
        X = X.drop(self.remove_cols_, axis=1)
        return X
    def get_feature_names_out(self):
        return [col for col in self.featurein_ if col not in self.remove_cols_]

3 Build Sub-Pipelines

from sklearn.pipeline import Pipeline
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import (
    StandardScaler, MinMaxScaler, OneHotEncoder)

# Numerical: เติม missing → scale
num_pipe = Pipeline([
    ("dropMissing", DropWMissing(threshold=0.5)),
    ('imputer', SimpleImputer(strategy='median')),
    ('scaler', StandardScaler())])
# StandardScaler: mean=0, std=1
# MinMaxScaler:   range [0,1]
# strategy: 'mean','median','most_frequent','constant'

# Categorical: เติม missing → encode
cat_pipe = Pipeline([
    ('imputer', SimpleImputer(strategy='most_frequent')),
    ('encoder', OneHotEncoder(handle_unknown='ignore',
                              sparse_output=False))])
# handle_unknown='ignore' → ค่าใหม่ใน test → 0
# sparse_output=False → return array ไม่ใช่ sparse

4 ColumnTransformer (รวม pipeline)

from sklearn.compose import ColumnTransformer

preprocessor = ColumnTransformer([
    ('num', num_pipe, num_features),
    ('cat', cat_pipe, cat_features)], remainder='passthrough')
preprocessor.set_output(transform='pandas') # ทำให้ output เป็น DataFrame และชื่อ feature จะเป็น ชื่อ pipe ย่อย + __ + featurename เช่น num__Age, cat__education_X (ถ้าไปผ่าน Onehot)

5 Transform Data

X_train_t = preprocessor.fit_transform(X_train)
X_test_t  = preprocessor.transform(X_test)
# fit_transform = เรียนรู้+แปลง (train เท่านั้น)
# transform = แปลงอย่างเดียว (test เท่านั้น)
fit บน train เท่านั้น ห้าม fit บน test → data leakage

6 Full Pipeline (preprocessor + model)

# เปลี่ยนเป็น make_pipeline โดยไม่ต้องใส่ list แต่เอา tuple transformer ต่อๆกันไม่ต้องใส่ชื่อ pipe ย่อย
full_pipe = Pipeline([
    ('preprocessor', preprocessor),
    ('model', RandomForestRegressor(random_state=42))])
